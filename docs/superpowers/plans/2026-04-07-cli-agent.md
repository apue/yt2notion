# CLI Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a file-backed CLI agent that queues URLs, processes them sequentially through the existing pipeline with `codex_cli + obsidian + summary`, and exposes progress/status commands.

**Architecture:** Add a dedicated agent runtime layer that owns `agent.yaml`, runtime `AGENTS.md`, queue files, job files, and worker orchestration. Reuse the existing pipeline for media processing, add a small progress callback hook for live job-state updates, and isolate Codex runtime behavior by passing an agent-home working directory to the Codex backend. Preserve existing `extract.asr.*` behavior from the repo `config.yaml` so remote ASR restart and healthcheck logic continue to work unchanged.

**Tech Stack:** Python 3.11+, Typer, subprocess, dataclasses, pathlib, JSON files, pytest

**Spec:** `docs/superpowers/specs/2026-04-07-cli-agent-design.md`

from __future__ import annotations

import json
from copy import deepcopy
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
import os

import yaml

from yt2notion.config import AppConfig, load_config


@dataclass(frozen=True)
class AgentConfig:
    vault_path: str
    summaries_dir: str
    transcripts_dir: str
    workspace_dir: str
    codex_model: str
    reasoning_effort: str


@dataclass(frozen=True)
class AgentPaths:
    home: Path
    config_path: Path
    agents_path: Path
    queue_path: Path
    worker_path: Path
    jobs_dir: Path
    logs_dir: Path
    workspace_dir: Path


def _now_iso() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


def default_runtime_agents_md() -> str:
    return "\n".join(
        [
            "# Runtime AGENTS.md",
            "",
            "- Codex is used only as a text-processing backend for media content.",
            "- summary mode only.",
            "- Obsidian writes are allowed only to the configured local vault.",
            "- No repository tasks, no code changes, no browsing, no autonomous agent behavior.",
            "- Always preserve source attribution in generated output.",
            "",
        ]
    )


def default_agent_yaml(home: Path) -> str:
    workspace_dir = home / "workspace"
    return "\n".join(
        [
            'vault_path: "/path/to/your/vault"',
            'summaries_dir: "yt2notion/summaries"',
            'transcripts_dir: "yt2notion/transcripts"',
            f'workspace_dir: "{workspace_dir}"',
            'codex_model: "gpt-5.3-codex"',
            'reasoning_effort: "low"',
            "",
        ]
    )


def ensure_agent_home(home: Path) -> AgentPaths:
    home.mkdir(parents=True, exist_ok=True)
    jobs_dir = home / "jobs"
    logs_dir = home / "logs"
    workspace_dir = home / "workspace"
    jobs_dir.mkdir(exist_ok=True)
    logs_dir.mkdir(exist_ok=True)
    workspace_dir.mkdir(exist_ok=True)
    config_path = home / "agent.yaml"
    agents_path = home / "AGENTS.md"
    queue_path = home / "queue.json"
    worker_path = home / "worker.json"
    if not config_path.exists():
        config_path.write_text(default_agent_yaml(home), encoding="utf-8")
    if not agents_path.exists():
        agents_path.write_text(default_runtime_agents_md(), encoding="utf-8")
    if not queue_path.exists():
        queue_path.write_text(
            json.dumps({"queued_job_ids": [], "updated_at": _now_iso()}, indent=2),
            encoding="utf-8",
        )
    return AgentPaths(
        home=home,
        config_path=config_path,
        agents_path=agents_path,
        queue_path=queue_path,
        worker_path=worker_path,
        jobs_dir=jobs_dir,
        logs_dir=logs_dir,
        workspace_dir=workspace_dir,
    )


def load_agent_config(paths: AgentPaths) -> AgentConfig:
    raw = yaml.safe_load(paths.config_path.read_text(encoding="utf-8")) or {}
    return AgentConfig(
        vault_path=str(raw["vault_path"]),
        summaries_dir=str(raw.get("summaries_dir", "yt2notion/summaries")),
        transcripts_dir=str(raw.get("transcripts_dir", "yt2notion/transcripts")),
        workspace_dir=os.path.expanduser(str(raw["workspace_dir"])),
        codex_model=str(raw.get("codex_model", "gpt-5.3-codex")),
        reasoning_effort=str(raw.get("reasoning_effort", "low")),
    )


def read_queue(paths: AgentPaths) -> dict:
    return json.loads(paths.queue_path.read_text(encoding="utf-8"))


def write_queue(paths: AgentPaths, queue: dict) -> None:
    queue["updated_at"] = _now_iso()
    paths.queue_path.write_text(json.dumps(queue, ensure_ascii=False, indent=2), encoding="utf-8")


def write_job(paths: AgentPaths, record: dict) -> None:
    job_path = paths.jobs_dir / f'{record["job_id"]}.json'
    paths.jobs_dir.mkdir(exist_ok=True)
    job_path.write_text(json.dumps(record, ensure_ascii=False, indent=2), encoding="utf-8")


def build_runtime_app_config(
    base_config_path: str,
    agent_config: AgentConfig,
    agent_home: Path,
) -> AppConfig:
    base = load_config(base_config_path)
    model = deepcopy(base.model)
    storage = deepcopy(base.storage)
    extract = deepcopy(base.extract)
    credit = deepcopy(base.credit)
    output = deepcopy(base.output)
    workspace = deepcopy(base.workspace)
    model["backend"] = "codex_cli"
    model["summarize_model"] = agent_config.codex_model
    model["translate_model"] = agent_config.codex_model
    model["review_model"] = agent_config.codex_model
    model["reasoning_effort"] = agent_config.reasoning_effort
    model["_runtime"] = {"codex_workdir": str(agent_home)}
    storage["backend"] = "obsidian"
    storage["obsidian"] = {
        "vault_path": agent_config.vault_path,
        "summaries_dir": agent_config.summaries_dir,
        "transcripts_dir": agent_config.transcripts_dir,
    }
    output["mode"] = "summary"
    workspace["base_dir"] = agent_config.workspace_dir
    return AppConfig(
        model=model,
        storage=storage,
        extract=extract,
        credit=credit,
        output=output,
        workspace=workspace,
    )
```

def _run_codex_exec(
    prompt: str,
    *,
    model: str,
    timeout_seconds: int,
    reasoning_effort: str,
    workdir: str | None,
) -> str:
    with tempfile.NamedTemporaryFile(prefix="yt2notion-codex-", suffix=".txt", delete=False) as f:
        output_path = Path(f.name)
    cmd = [
        "codex",
        "exec",
        "-m",
        model,
        "-c",
        f'model_reasoning_effort="{reasoning_effort}"',
        "--sandbox",
        "read-only",
        "--output-last-message",
        str(output_path),
        "-",
    ]
    try:
        subprocess.run(
            cmd,
            input=prompt,
            capture_output=True,
            text=True,
            check=True,
            timeout=timeout_seconds,
            cwd=workdir,
        )
        raw = output_path.read_text(encoding="utf-8")
    finally:
        output_path.unlink(missing_ok=True)
    text = raw.strip()
    if not text:
        raise _EmptyOutputError("codex returned empty output")
    return text


class CodexCLICaller:
    def __init__(
        self,
        model: str = "gpt-5.2",
        *,
        timeout_seconds: int = 300,
        reasoning_effort: str = "low",
        workdir: str | None = None,
    ) -> None:
        self.model = _normalize_codex_model(model)
        self.timeout_seconds = timeout_seconds
        self.reasoning_effort = _normalize_reasoning_effort(reasoning_effort)
        self.workdir = workdir

    def call(self, system_prompt: str, user_prompt: str, *, max_tokens: int = 4000) -> str:
        del max_tokens
        prompt = f"{system_prompt}\n\n---\n\n{user_prompt}"
        return _run_codex_exec(
            prompt,
            model=self.model,
            timeout_seconds=self.timeout_seconds,
            reasoning_effort=self.reasoning_effort,
            workdir=self.workdir,
        )

from typing import Callable

ProgressCallback = Callable[[str, str, str | None], None]


def _emit_progress(
    callback: ProgressCallback | None,
    step: str,
    event: str,
    message: str | None = None,
) -> None:
    if callback is not None:
        callback(step, event, message)


def run_pipeline(
    url: str,
    config: AppConfig,
    *,
    verbose: bool = False,
    dry_run: bool = False,
    resume_from: str | None = None,
    workspace_dir: str | None = None,
    mode: str | None = None,
    progress_callback: ProgressCallback | None = None,
) -> str:
    prepared = prepare_content(
        url,
        config,
        verbose=verbose,
        resume_from=resume_from,
        workspace_dir=workspace_dir,
        mode=mode,
        progress_callback=progress_callback,
    )
    _emit_progress(progress_callback, "publish", "started")
    result_url = storage.save(
        prepared.chinese_content,
        prepared.metadata,
        transcript_segments=transcript_segments,
        entities=prepared.entities,
    )
    _emit_progress(progress_callback, "publish", "completed", result_url)


def prepare_content(
    url: str,
    config: AppConfig,
    *,
    verbose: bool = False,
    resume_from: str | None = None,
    workspace_dir: str | None = None,
    mode: str | None = None,
    progress_callback: ProgressCallback | None = None,
) -> PreparedContent:
    _emit_progress(progress_callback, "download", "started")
    metadata = _step_download(url, raw_config, verbose)
    _emit_progress(progress_callback, "download", "completed")
    _emit_progress(progress_callback, "segment", "started")
    segments = _step_segment(metadata, raw_config, verbose)
    _emit_progress(progress_callback, "segment", "completed")
    _emit_progress(progress_callback, "transcribe", "started")
    transcripts = _step_transcribe(ws, metadata, segments, raw_config, verbose)
    _emit_progress(progress_callback, "transcribe", "completed")
    _emit_progress(progress_callback, "extract", "started")
    entities = _step_extract(analysis_segments, raw_config, verbose)
    _emit_progress(progress_callback, "extract", "completed")
    _emit_progress(progress_callback, "summarize", "started")
    chinese_content = _step_summarize(
        analysis_segments,
        metadata,
        raw_config,
        output_mode=output_mode,
        is_long=is_long,
        verbose=verbose,
    )
    _emit_progress(progress_callback, "summarize", "completed")

from __future__ import annotations

import json
import os
import signal
import subprocess
import sys
from datetime import datetime
from pathlib import Path

from yt2notion.agent_runtime import (
    AgentConfig,
    AgentPaths,
    build_runtime_app_config,
    read_queue,
    write_job,
    write_queue,
)
from yt2notion.pipeline import run_pipeline


def _now_iso() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


def _read_job(paths: AgentPaths, job_id: str) -> dict:
    return json.loads((paths.jobs_dir / f"{job_id}.json").read_text(encoding="utf-8"))


def _job_log_path(paths: AgentPaths, job_id: str) -> Path:
    return paths.logs_dir / f"{job_id}.log"


def append_job_log(paths: AgentPaths, job_id: str, line: str) -> None:
    with _job_log_path(paths, job_id).open("a", encoding="utf-8") as fh:
        fh.write(f"{_now_iso()} {line}\n")


def _pid_alive(pid: int) -> bool:
    try:
        os.kill(pid, 0)
    except OSError:
        return False
    return True


def mark_stale_worker_failed(paths: AgentPaths) -> None:
    if not paths.worker_path.exists():
        return
    worker = json.loads(paths.worker_path.read_text(encoding="utf-8"))
    pid = int(worker.get("pid", 0))
    if pid and _pid_alive(pid):
        return
    job_id = worker.get("job_id")
    if job_id:
        record = _read_job(paths, job_id)
        if record["status"] in {"starting", "running"}:
            record["status"] = "failed"
            record["error"] = "worker process exited before job completion"
            record["finished_at"] = _now_iso()
            record["updated_at"] = _now_iso()
            write_job(paths, record)
    paths.worker_path.unlink(missing_ok=True)


def retry_job(paths: AgentPaths, job_id: str) -> str:
    original = _read_job(paths, job_id)
    if original["status"] != "failed":
        raise ValueError("Only failed jobs can be retried")
    new_job_id = f'{datetime.now().strftime("%Y%m%d-%H%M%S")}-retry'
    new_record = {
        "job_id": new_job_id,
        "url": original["url"],
        "status": "queued",
        "created_at": _now_iso(),
        "updated_at": _now_iso(),
        "started_at": None,
        "finished_at": None,
        "current_step": None,
        "completed_steps": [],
        "workspace_dir": None,
        "video_id": None,
        "title": None,
        "channel": None,
        "result_path": None,
        "error": None,
    }
    write_job(paths, new_record)
    queue = read_queue(paths)
    queue["queued_job_ids"].append(new_job_id)
    write_queue(paths, queue)
    return new_job_id


def run_worker_once(paths: AgentPaths, agent_config: AgentConfig, *, base_config_path: str) -> bool:
    queue = read_queue(paths)
    if not queue["queued_job_ids"]:
        paths.worker_path.unlink(missing_ok=True)
        return False
    job_id = queue["queued_job_ids"].pop(0)
    write_queue(paths, queue)
    record = _read_job(paths, job_id)
    record["status"] = "starting"
    record["started_at"] = _now_iso()
    record["updated_at"] = _now_iso()
    write_job(paths, record)
    paths.worker_path.write_text(
        json.dumps(
            {"pid": os.getpid(), "job_id": job_id, "started_at": _now_iso(), "mode": "background"},
            ensure_ascii=False,
            indent=2,
        ),
        encoding="utf-8",
    )

    app_config = build_runtime_app_config(base_config_path, agent_config, paths.home)

    def on_progress(step: str, event: str, message: str | None = None) -> None:
        current = _read_job(paths, job_id)
        current["updated_at"] = _now_iso()
        if event == "started":
            current["status"] = "running"
            current["current_step"] = step
        elif event == "completed":
            current["completed_steps"] = list(dict.fromkeys([*current["completed_steps"], step]))
        write_job(paths, current)
        append_job_log(paths, job_id, f"{step}:{event} {message or ''}".strip())

    try:
        result_path = run_pipeline(
            record["url"],
            app_config,
            verbose=True,
            mode="summary",
            progress_callback=on_progress,
        )
    except Exception as exc:
        failed = _read_job(paths, job_id)
        failed["status"] = "failed"
        failed["error"] = str(exc)
        failed["finished_at"] = _now_iso()
        failed["updated_at"] = _now_iso()
        write_job(paths, failed)
        append_job_log(paths, job_id, f"FAILED {exc}")
        paths.worker_path.unlink(missing_ok=True)
        return True

    completed = _read_job(paths, job_id)
    completed["status"] = "completed"
    completed["result_path"] = result_path
    completed["finished_at"] = _now_iso()
    completed["updated_at"] = _now_iso()
    write_job(paths, completed)
    append_job_log(paths, job_id, f"COMPLETED {result_path}")
    paths.worker_path.unlink(missing_ok=True)
    return True


def spawn_background_worker(paths: AgentPaths, *, base_config_path: str) -> subprocess.Popen:
    stdout_path = paths.logs_dir / "worker.log"
    stdout_handle = stdout_path.open("a", encoding="utf-8")
    return subprocess.Popen(
        [
            sys.executable,
            "-m",
            "yt2notion.cli",
            "agent",
            "_worker",
            "--agent-home",
            str(paths.home),
            "--config",
            base_config_path,
        ],
        cwd=str(paths.home),
        stdout=stdout_handle,
        stderr=subprocess.STDOUT,
        start_new_session=True,
        text=True,
    )
```

@agent_app.command("run")
def agent_run(

    agent_home: str = typer.Option("~/.yt2notion-agent", "--agent-home"),
